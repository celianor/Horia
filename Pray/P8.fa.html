⌂ خانه ↩ بازگشت
در آستانه‌ی خاموشی

نیایشِ سکوت

✦
دعایی برای بازگشتن به خویش

ای هُورا،

ای آن‌که نامت را
زبان توانِ گفتن ندارد
و حضورت را
دل توانِ پنهان کردن؛

ای نزدیکِ دور،
ای آشکارِ پنهان،
ای نادیده‌ی دیدنی،
ای نشنیده‌ی شنیدنی؛

تو که گاه
در یک نفسِ آرام
از هزاران سخن
به من نزدیک‌تری، امشب از تو
نه پاسخ می‌خواهم،
نه راه،
نه رهایی؛

فقط
اندکی سکوت.

نه سکوتِ لب،
که سکوتِ دل؛
نه خاموشیِ صدا،
که آرامشِ معنا؛
نه گریختن از جهان،
که بازگشتن به خویش.

مرا لحظه‌ای
از این همه گفتن رها کن؛
از گفتنِ آنچه می‌دانم،
از گفتنِ آنچه نمی‌دانم،
از گفتن برای آنکه دیده شوم،
از گفتن برای آنکه شنیده شوم،
از گفتن برای آنکه حق با من باشد.

مرا از این «من»
کمی دور کن؛
از این منِ کوچک
که پیوسته می‌خواهد،
می‌ترسد،
می‌سنجد،
می‌رَنجد،
می‌خواهد دیده شود
و جهان را
گردِ خویش می‌گرداند.

شاید بزرگ‌ترین هیاهوی جهان
نه در کوچه‌هاست،
نه در بازارها،
نه در شهرها؛
بلکه در درونِ من است؛
در ازدحامِ خواهش‌ها،
در غوغای خاطره‌ها،
در کشاکشِ ترس و تمنا.

مرا از این هیاهو
به ساحلِ سکوت ببر؛

آنجا که لازم نیست
ثابت کنم کیستم؛
آنجا که لازم نیست
برای بودن،
دلیلی بیاورم؛
آنجا که می‌توانم
فقط باشم:

بی‌نام،
بی‌نقاب،
بی‌ادعا؛

چون برگ که برگ است،
و باران که باران؛
بی‌آنکه خویش را
برای جهان توضیح دهند.

ای هُورا،

به من بیاموز
که هر سخنی
پاسخی نمی‌خواهد؛
هر زخمی
انتقامی نمی‌خواهد؛
هر رفتنی
دنبال‌کردنی نیست؛
و هر سکوتی
نشانه‌ی شکست نیست.

گاهی سکوت
بلوغِ سخن است؛
گاهی درنگ
نجاتِ سخن؛
و گاهی نگفتن
صادقانه‌ترین گفتن.

گاهی
کنارِ دلی شکسته نشستن
و هیچ نگفتن،
از هزار وعده
راست‌تر است.

گاهی
دست بر شانه‌ای گذاشتن
و خاموش ماندن،
از هزار «دوستت دارم»
دوست‌داشتنی‌تر است.

پس اگر روزی
کسی با دلی شکسته
به من پناه آورد،
مرا از سخنِ آماده خالی کن.

مگذار فوراً
دردش را معنا کنم؛
مگذار اندوهش را
با جمله‌های زیبا
کوچک کنم؛
مگذار برای زخمش
پاسخی بسازم
پیش از آنکه
زخم او را ببینم.

فقط بیاموز
کنارش بمانم،
بشنوم،
و اگر لازم شد
دستش را بگیرم.

ای هُورا،

مرا بزرگ مکن؛
عمیق کن.

نامم را بلند مکن؛
قدمم را استوار کن.

مرا مشهور مکن؛
مفید کن.

مرا بر سرِ زبان‌ها مینداز؛
بر سرِ راهِ رنج‌دیدگان بگذار.

که شاید ارزشِ انسان
به مقدارِ نوری نباشد
که بر او می‌تابد؛
به مقدارِ نوری باشد
که در تاریکیِ دیگری
می‌افروزد.

به من سکوتی بده
که مرا از آدمیان جدا نکند،
بلکه به آنان نزدیک‌تر کند؛

سکوتی
که از بی‌مهری نیست،
از فهم است؛
از شتاب نیست،
از حضور است؛
از ناتوانی نیست،
از توانِ خویشتن‌داری است.

و مرا از سکوتِ سنگینِ کینه
نگه دار؛
از آن لب‌های بسته
که دل‌هایشان هنوز در جنگ است؛
از آن آرامشِ ظاهری
که زیرِ خاکسترش
آتشِ انتقام نفس می‌کشد.

خاموشی‌ای ببخش
که در آن آتش فروکش کند،
نه آنکه پنهان بماند؛
درد آرام گیرد،
نه آنکه در تاریکی
ریشه بدواند.

اگر رنجیده‌ام،
مرا به فهم برسان؛
اگر خشمگینم،
به درنگ؛
اگر شکسته‌ام،
به پذیرش؛
و اگر بخشیدم،
از منت نجاتم ده.

که بخششِ حقیقی
آن نیست که دیگری را
از خطایش برهانیم؛
آن است که خود را
از زنجیرِ خطای او آزاد کنیم.

شاید آزادی
همین باشد:

که گذشته
دیگر نتواند
امروزِ ما را زندانی کند؛

که رفتنِ کسی
پایانِ بودنِ ما نباشد؛

که زخم
تمامِ هویتِ ما نشود؛

و بتوانیم
با تمامِ زخم‌ها
باز هم
دوست بداریم؛

نه از سرِ فراموشی،
از سرِ فهم.

ای خدایِ جان،

مرا از خودم نجات بده؛
اما نه با نابود کردنِ «من»؛
با بزرگ‌تر کردنِ آن.

آن‌قدر بزرگش کن
که دیگر
تنها خود را نبیند؛

که در خود،
دیگری را ببیند؛

در دیگری،
جهان را؛

و در جهان،
رازِ بودن را.

شاید انسان
آنگاه بالغ می‌شود
که می‌فهمد
جهان برای او آفریده نشده است؛

و عمیق‌تر از آن،
آنگاه که می‌فهمد
او نیز برای خویشتن
نیامده است.

ما آمده‌ایم
تا چیزی بیفزاییم:

اندکی مهر،
اندکی روشنایی،
اندکی آرامش؛

تا جهان
پس از عبورِ ما
کمی مهربان‌تر باشد
از جهانی که پیش از ما بود.

اگر قدرت دادی،
پناه باشم،
نه پناه‌گیر؛

اگر دانایی دادی،
فروتن باشم،
نه فخرفروش؛

اگر توانایی دادی،
یاری باشم،
نه سلطه؛

اگر ثروت دادی،
ببخشم،
نه آنکه تنها بیندوزم.

و اگر تنهایم گذاشتی،
در آن تنهایی
چیزی نشانم ده
که در ازدحام
هرگز نمی‌دیدم.

زیرا شاید تنهایی
نبودنِ دیگران نیست؛
فرصتِ دیدنِ خویشتن است.

و شاید خلوت
جای خالیِ آدم‌ها نیست؛
جایی است
که حقیقت
سرانجام فرصت می‌کند
آهسته سخن بگوید.

ای هُورا،

اگر در آن خلوت
از خود پرسیدم:

«من کیستم؟»

مرا به پاسخ‌های آسان
قانع مکن.

مگذار بگویم
من نامم هستم،
شغلم هستم،
دارایی‌ام هستم،
گذشته‌ام هستم،
یا تصویری که دیگران
از من ساخته‌اند.

مرا ژرف‌تر ببر؛
آن‌قدر ژرف
که بفهمم
هیچ‌یک از این‌ها
تمامِ من نیست.

و همان‌جا
که هنوز
نامی برای خویش ندارم،
مرا نگه دار.

زیرا شاید
آغازِ حکمت
از همان‌جاست
که انسان
جرئت می‌کند بگوید:

«نمی‌دانم.»

ای هُورا،

به من شجاعتِ ندانستن بده؛
دانشی که فروتنم کند؛
فهمی که مهربانم کند؛
و حقیقتی که
مرا از حقیقتِ دیگری
غافل نکند.

اگر بسیار بدانم
و دردِ یک انسان را نفهمم،
دانایی‌ام
چه ارزشی دارد؟

اگر جهان را بشناسم
و دلِ خویش را نه،
این همه دانستن
چه سود؟

به من زبانی بده
که راست بگوید
و راست‌گویی‌اش سنگ نباشد.

بیاموز
که حقیقت و محبت
دشمن یکدیگر نیستند؛

می‌توان راست گفت
و مهربان ماند؛

می‌توان مخالف بود
و دشمن نبود؛

می‌توان رفت
و کینه نبرد؛

می‌توان بخشید
و فراموش نکرد؛

و می‌توان فراموش کرد
بی‌آنکه حقیقت را انکار کرد.

ای هُورا،

اکنون
صدای باران را می‌شنوم.

برگ می‌لرزد.

شب نفس می‌کشد.

چراغی دور
در پنجره‌ای روشن است.

و ناگهان
چیزی در من آرام می‌شود.

شاید جهان
همین‌قدر ساده است؛
و ما
آن را با خواستن‌های بسیار
پیچیده کرده‌ایم.

شاید حقیقت
دور نیست.

شاید
در همین نفس است؛
در همین نگاه؛
در همین لحظه؛
در همین فرصتِ کوتاه
برای دوست داشتن.

ای هُورا،

امشب از تو
جاودانگی نمی‌خواهم.

فقط
مرا شایسته‌ی همین لحظه کن.

اگر خنده‌ای هست،
از تهِ دل باشد؛

اگر اشکی هست،
پاک باشد؛

اگر عشقی هست،
آزاد باشد؛

اگر سکوتی هست،
عمیق باشد؛

و اگر سخنی هست،
به شنیدن بیارزد.

زیرا اکنون می‌دانم:

سکوت
پایانِ صدا نیست؛
آغازِ شنیدن است.

عشق
داشتنِ دیگری نیست؛
آزاد گذاشتنِ اوست.

دانایی
بسیار دانستن نیست؛
بیشتر فهمیدن است.

و زندگی
بسیار زیستن نیست؛
عمیق زیستن است.

پس اگر قرار است
مرا تغییر دهی،
از زبانم آغاز نکن؛
از نگاهم آغاز کن.

نگاهم را چنان عوض کن
که همان جهان را
دیگرگونه ببینم.

و اگر قرار است
سکوت کنم،
چنان خاموشم کن
که در خاموشی
جهان را بیشتر بشنوم.

تا روزی برسد
که دیگر نپرسم:

«تو کجایی؟»

زیرا خواهم دانست
تو را باید در نشانه‌ها جست:

در آن نفسی
که بخشیدم؛

در آن دستی
که گرفتم؛

در آن اشکی
که پاک کردم؛

در آن سخنی
که نگفتم؛

در آن بخششی
که منت نگذاشتم؛

و در آن سکوتی
که برای شنیدن
برگزیدم.

آنجا تو را خواهم یافت:

نه در دوردست،
که در ژرفا؛

نه در هیاهو،
که در حضور؛

نه تنها در پاسخ،
که در پرسش؛

نه در پایان،
که در راه.

ای هُورا،

اکنون لب می‌بندم؛

نه از آن رو
که دیگر چیزی برای گفتن نیست؛
از آن رو
که بعضی حقیقت‌ها
وقتی به زبان می‌آیند،
کوچک می‌شوند.

بگذار
آخرین سخن را
سکوت بگوید.

و اگر از این سکوت
صدایی شنیده شد،
آن را صدای من مپندار.

شاید صدای همان حقیقتی باشد
که سال‌ها
در من خاموش بود
و اکنون
آرام،
بی‌شتاب،
بی‌ادعا
مرا به خویش می‌خواند.

و من می‌روم؛

نه با فریاد،
با سکوت؛

نه با یقینِ مغرور،
با حیرتِ روشن؛

نه با ادعای دانستن،
با شوقِ فهمیدن؛

و نه برای آنکه
از جهان بگریزم،

برای آنکه
به جهان
با قلبی بیدارتر
بازگردم.

که شاید
مقصدِ خاموشی
گریختن از جهان نباشد؛

آموختنِ چگونه زیستن در جهان باشد.

پس ای هُورا،

اگر مرا خاموش کردی،
شنواترم کن؛

اگر مرا تنها کردی،
عمیق‌ترم کن؛

اگر مرا شکستی،
مهربان‌ترم کن؛

و اگر مرا به خویش بازگرداندی،
مگذار
باز همان آدمِ پیشین باشم.

مرا چنان بازگردان
که هرکه مرا دید،
اندکی آرام‌تر شود؛
هرکه با من سخن گفت،
اندکی سبک‌تر شود؛
و هرکه از کنارم گذشت،
حتی اگر نامم را ندانست،
نشانی از مهر
با خویش ببرد.

این است دعای من.

نه بیشتر.

نه کمتر.

که اگر نتوانم
جهان را روشن کنم،
دست‌کم
چراغی در دلِ یک نفر
خاموش نکنم.

اگر نتوانم
تمامِ رنجِ جهان را درمان کنم،
دست‌کم
بر رنجِ کسی
رنجی نیفزایم.

و اگر نتوانم
رازِ هستی را دریابم،
دست‌کم
رازِ انسان بودن را
فراموش نکنم:

که آمدن،
برای گرفتن نیست؛
برای بخشیدن است.

که دانستن،
برای فخر نیست؛
برای فهمیدن است.

که توانستن،
برای سلطه نیست؛
برای پناه بودن است.

و که دوست داشتن،
برای داشتن نیست؛
برای آزاد گذاشتن است.

ای هُورا،

اکنون
دیگر چیزی نمی‌گویم.

لب می‌بندم.

دل می‌گشایم.

نفس می‌کشم.

و گوش می‌سپارم.

به باران؛
به برگ؛
به شب؛
به نفس؛
به سکوت.

و شاید
در ژرفای همین سکوت
بشنوم
آنچه سال‌ها
در هیاهو نشنیدم:

که تو
هیچ‌گاه دور نبوده‌ای.

من دور بوده‌ام.

و اکنون
هرچه از هیاهو کم می‌کنم،
تو را نزدیک‌تر می‌یابم؛

هرچه از «من» می‌کاهم،
جهان را بیشتر می‌بینم؛

و هرچه بیشتر می‌بخشم،
خویشتن را بیشتر می‌یابم.

پس بگذار
در این واپسین لحظه
هیچ نگویم.

زیرا شاید
تمامِ نیایش
همین باشد:

خاموش شدنِ من
و شنیده شدنِ تو.

آمین.

❦
در سکوت، آنچه پنهان بود شنیده می‌شود.
